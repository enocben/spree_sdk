Future<Address> create({
  required AddressParams params,
  RequestOptions? options,
}) async {
  final Object? json = await client.request(
    'POST',
    '/customers/me/addresses',
    body: params.toJson(),
    options: options,
  );
  return Address.fromJson(_asObject(json, 'addresses.create'));
}
