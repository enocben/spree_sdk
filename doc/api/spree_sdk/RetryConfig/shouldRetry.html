Whether a failed attempt qualifies for retry number retryNumber (1-based: 1 is the first retry after the initial attempt).

networkError covers transport/timeout failures with no response. idempotent must be true for non-safe methods (explicit idempotency key); safe methods are always idempotent.

bool shouldRetry({
  required String method,
  required int retryNumber,
  int? statusCode,
  bool networkError = false,
  bool idempotent = false,
}) {
  if (retryNumber < 1 || retryNumber > maxRetries) {
    return false;
  }
  final bool safe = safeHttpMethods.contains(method.toUpperCase());
  if (!safe && !idempotent) {
    return false;
  }
  if (networkError) {
    return true;
  }
  if (statusCode == null) {
    return false;
  }
  return retryableStatuses.contains(statusCode);
}
