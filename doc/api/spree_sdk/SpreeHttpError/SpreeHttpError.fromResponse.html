Builds a SpreeHttpError — or a SpreeAuthError for 401/403 — from a failed response. Never throws: unparseable bodies yield a status-based message instead.

factory SpreeHttpError.fromResponse({
  required String method,
  required Uri uri,
  required int statusCode,
  required String bodyText,
}) {
  String? code;
  String message = 'Request failed with status $statusCode';
  Object? details;
  String? rawBody;
  if (bodyText.isNotEmpty) {
    rawBody = bodyText.length > 2000 ? bodyText.substring(0, 2000) : bodyText;
    try {
      final Object? decoded = jsonDecode(bodyText);
      if (decoded is Map) {
        final Object? error = decoded['error'];
        if (error is Map) {
          final Object? rawCode = error['code'];
          if (rawCode != null) {
            code = rawCode.toString();
          }
          final Object? rawMessage = error['message'];
          if (rawMessage is String && rawMessage.isNotEmpty) {
            message = rawMessage;
          }
          details = error['details'];
        }
      }
    } on FormatException {
      // Keep the status-based message; rawBody retains diagnostics.
    }
  }
  if (statusCode == 401 || statusCode == 403) {
    return SpreeAuthError(
      statusCode: statusCode,
      message: message,
      method: method,
      uri: uri,
      code: code,
      details: details,
      rawBody: rawBody,
    );
  }
  return SpreeHttpError(
    statusCode: statusCode,
    message: message,
    method: method,
    uri: uri,
    code: code,
    details: details,
    rawBody: rawBody,
  );
}
